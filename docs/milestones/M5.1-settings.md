### 2. SettingsStore.swift

```swift
// SayVoice/Settings/SettingsStore.swift
import Foundation
import Observation

@MainActor @Observable
final class SettingsStore {
    var hotkeyCode: Int {
        didSet { UserDefaults.standard.set(hotkeyCode, forKey: SettingsKeys.hotkeyCode) }
    }
    var modelSize: String {
        didSet { UserDefaults.standard.set(modelSize, forKey: SettingsKeys.modelSize) }
    }
    var language: String {
        didSet { UserDefaults.standard.set(language, forKey: SettingsKeys.language) }
    }
    var overlayEnabled: Bool {
        didSet { UserDefaults.standard.set(overlayEnabled, forKey: SettingsKeys.overlayEnabled) }
    }
    var soundFeedback: Bool {
        didSet { UserDefaults.standard.set(soundFeedback, forKey: SettingsKeys.soundFeedback) }
    }
    var pasteMethod: String {
        didSet { UserDefaults.standard.set(pasteMethod, forKey: SettingsKeys.pasteMethod) }
    }
    var launchAtLogin: Bool {
        didSet { UserDefaults.standard.set(launchAtLogin, forKey: SettingsKeys.launchAtLogin) }
    }
    var hasCompletedOnboarding: Bool {
        didSet { UserDefaults.standard.set(hasCompletedOnboarding, forKey: SettingsKeys.hasCompletedOnboarding) }
    }

    init() {
        let d = UserDefaults.standard
        self.hotkeyCode           = d.object(forKey: SettingsKeys.hotkeyCode) as? Int ?? 61
        self.modelSize            = d.string(forKey: SettingsKeys.modelSize) ?? "small"
        self.language             = d.string(forKey: SettingsKeys.language) ?? "auto"
        self.overlayEnabled       = d.object(forKey: SettingsKeys.overlayEnabled) as? Bool ?? true
        self.soundFeedback        = d.object(forKey: SettingsKeys.soundFeedback) as? Bool ?? true
        self.pasteMethod          = d.string(forKey: SettingsKeys.pasteMethod) ?? "auto"
        self.launchAtLogin        = d.bool(forKey: SettingsKeys.launchAtLogin)
        self.hasCompletedOnboarding = d.bool(forKey: SettingsKeys.hasCompletedOnboarding)
    }
}
```

**Заметки:**
- `@MainActor @Observable` — не `ObservableObject`, macOS 14+ паттерн
- `object(forKey:) as? Bool ?? true` для overlayEnabled/soundFeedback — иначе `bool(forKey:)` вернёт `false` при отсутствии ключа
- Defaults: hotkeyCode=61 (Right Option), modelSize="small", language="auto"

### 3. SettingsView.swift

```swift
// SayVoice/UI/SettingsView.swift
import SwiftUI

struct SettingsView: View {
    @Bindable var settings: SettingsStore

    var body: some View {
        Form {
            Section("Хоткей") {
                LabeledContent("Хоткей записи") {
                    Text(hotkeyLabel)
                        .foregroundStyle(.secondary)
                }
                Text("Настройка хоткея будет добавлена позже")
                    .font(.caption).foregroundStyle(.tertiary)
            }

            Section("Модель распознавания") {
                Picker("Модель", selection: $settings.modelSize) {
                    Text("Tiny (75 MB, быстрая)").tag("tiny")
                    Text("Base (142 MB, баланс)").tag("base")
                    Text("Small (465 MB, рекомендуется)").tag("small")
                }
                .pickerStyle(.radioGroup)
            }

            Section("Язык распознавания") {
                Picker("Язык", selection: $settings.language) {
                    Text("Авто (RU/EN)").tag("auto")
                    Text("Русский").tag("ru")
                    Text("English").tag("en")
                }
                .pickerStyle(.radioGroup)
            }

            Section("Интерфейс") {
                Toggle("Показывать overlay при записи", isOn: $settings.overlayEnabled)
                Toggle("Звуковая обратная связь", isOn: $settings.soundFeedback)
            }

            Section("Метод вставки текста") {
                Picker("Метод", selection: $settings.pasteMethod) {
                    Text("Авто (AX → Буфер обмена)").tag("auto")
                    Text("Только Accessibility").tag("ax")
                    Text("Только буфер обмена").tag("pasteboard")
                }
                .pickerStyle(.radioGroup)
            }

            Section("Система") {
                Toggle("Запускать при входе в систему", isOn: $settings.launchAtLogin)
            }
        }
        .formStyle(.grouped)
        .frame(width: 480, height: 520)
    }

    private var hotkeyLabel: String {
        settings.hotkeyCode == 61 ? "Right Option (⌥)" : "KeyCode: \(settings.hotkeyCode)"
    }
}
```

**Заметки:**
- `@Bindable var settings` — macOS 14+ паттерн для @Observable (не @EnvironmentObject)
- Хоткей: пока только отображение, кастомизация отложена
- Launch at login toggle: UI готов, SMAppService подключается в M5.5
